const SHEET_NAME = "Orders";

function setupSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
  }
  if (sheet.getLastRow() === 0) {
    sheet.appendRow([
      "Date & Time",
      "Invoice No",
      "Created By",
      "Customer Name",
      "Contact Number",
      "Company Name",
      "Aadhaar / GST No",
      "Delivery Address",
      "Products & Quantity",
      "Total Items",
      "Payment Status",
      "Packing Status",
      "Packed Checklist",
      "Box / Weight",
      "Courier / Transport",
      "Tracking / AWB / LR",
      "Remarks",
      "Order Image",
      "Packing Image"
    ]);
    sheet.getRange(1, 1, 1, 19).setFontWeight("bold");
  }
  return sheet;
}

function doGet(e) {
  const sheet = setupSheet();
  const data = sheet.getDataRange().getValues();
  const orders = [];
  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    if (!row[3]) continue;
    orders.push({
      createdAt: String(row[0] || ""),
      invoiceNo: String(row[1] || ""),
      createdBy: String(row[2] || ""),
      customerName: String(row[3] || ""),
      contactNo: String(row[4] || ""),
      companyName: String(row[5] || ""),
      taxId: String(row[6] || ""),
      address: String(row[7] || ""),
      items: String(row[8] || ""),
      itemCount: Number(row[9] || 1),
      paymentStatus: String(row[10] || "Success"),
      packingStatus: String(row[11] || "Pending"),
      packedFlags: String(row[12] || ""),
      weightBox: String(row[13] || ""),
      courierName: String(row[14] || ""),
      trackingId: String(row[15] || ""),
      remarks: String(row[16] || ""),
      orderImage: String(row[17] || ""),
      packingImage: String(row[18] || "")
    });
  }
  return ContentService.createTextOutput(JSON.stringify({ status: "ok", orders: orders.reverse() }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const sheet = setupSheet();
  const payload = JSON.parse(e.postData.contents);
  const data = sheet.getDataRange().getValues();

  if (payload.action === "create") {
    const o = payload.order;
    sheet.appendRow([
      o.createdAt,
      o.invoiceNo || "",
      o.createdBy || "",
      o.customerName,
      o.contactNo,
      o.companyName || "",
      o.taxId || "",
      o.address,
      o.items,
      o.itemCount || 1,
      "Success",
      o.packingStatus || "Pending",
      o.packedFlags || "",
      "",
      "",
      "",
      o.remarks || "",
      o.orderImage || "",
      o.packingImage || ""
    ]);
    return ContentService.createTextOutput(JSON.stringify({ status: "ok" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  if (payload.action === "delete") {
    for (let i = 1; i < data.length; i++) {
      const matchTime = String(data[i][0]) === String(payload.createdAt);
      const matchName = String(data[i][3]) === String(payload.customerName);
      if (matchTime && matchName) {
        sheet.deleteRow(i + 1);
        break;
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ status: "ok" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  for (let i = 1; i < data.length; i++) {
    const matchTime = String(data[i][0]) === String(payload.createdAt);
    const matchName = String(data[i][3]) === String(payload.matchCustomerName || payload.customerName);
    if (matchTime && matchName) {
      const rowNum = i + 1;
      if (payload.action === "editOrder") {
        const o = payload.order;
        sheet.getRange(rowNum, 4).setValue(o.customerName);
        sheet.getRange(rowNum, 5).setValue(o.contactNo);
        sheet.getRange(rowNum, 6).setValue(o.companyName || "");
        sheet.getRange(rowNum, 7).setValue(o.taxId || "");
        sheet.getRange(rowNum, 8).setValue(o.address);
        sheet.getRange(rowNum, 9).setValue(o.items);
        sheet.getRange(rowNum, 10).setValue(o.itemCount || 1);
        sheet.getRange(rowNum, 13).setValue(o.packedFlags || "");
        if (o.orderImage) sheet.getRange(rowNum, 18).setValue(o.orderImage);
      } else if (payload.action === "update") {
        sheet.getRange(rowNum, 12).setValue(payload.packingStatus);
        sheet.getRange(rowNum, 13).setValue(payload.packedFlags || "");
        sheet.getRange(rowNum, 14).setValue(payload.weightBox || "");
        sheet.getRange(rowNum, 15).setValue(payload.courierName || "");
        sheet.getRange(rowNum, 16).setValue(payload.trackingId || "");
        sheet.getRange(rowNum, 17).setValue(payload.remarks || "");
        if (payload.packingImage) sheet.getRange(rowNum, 19).setValue(payload.packingImage);
      }
      break;
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ status: "ok" }))
    .setMimeType(ContentService.MimeType.JSON);
}
