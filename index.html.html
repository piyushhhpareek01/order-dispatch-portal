const SHEET_NAME = "Orders";

function setupSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
  }
  if (sheet.getLastRow() === 0) {
    sheet.appendRow([
      "Date & Time",
      "Customer Name",
      "Contact No",
      "Company Name",
      "Aadhaar / GST No",
      "Address",
      "Items & Quantity",
      "Total Items",
      "Payment Status",
      "Packing Status",
      "Packed Checklist",
      "Weight / Box Size",
      "Courier Name",
      "Tracking ID / AWB",
      "Remarks"
    ]);
    sheet.getRange(1, 1, 1, 15).setFontWeight("bold");
  }
  return sheet;
}

function doGet(e) {
  const sheet = setupSheet();
  const data = sheet.getDataRange().getValues();
  const orders = [];
  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    if (!row[1]) continue;
    orders.push({
      createdAt: String(row[0] || ""),
      customerName: String(row[1] || ""),
      contactNo: String(row[2] || ""),
      companyName: String(row[3] || ""),
      taxId: String(row[4] || ""),
      address: String(row[5] || ""),
      items: String(row[6] || ""),
      itemCount: Number(row[7] || 1),
      paymentStatus: String(row[8] || "Success"),
      packingStatus: String(row[9] || "Pending"),
      packedFlags: String(row[10] || ""),
      weightBox: String(row[11] || ""),
      courierName: String(row[12] || ""),
      trackingId: String(row[13] || ""),
      remarks: String(row[14] || "")
    });
  }
  return ContentService.createTextOutput(JSON.stringify({ status: "ok", orders: orders.reverse() }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const sheet = setupSheet();
  const payload = JSON.parse(e.postData.contents);
  const data = sheet.getDataRange().getValues();

  if (payload.action === "create") {
    const o = payload.order;
    sheet.appendRow([
      o.createdAt,
      o.customerName,
      o.contactNo,
      o.companyName || "",
      o.taxId || "",
      o.address,
      o.items,
      o.itemCount || 1,
      "Success",
      o.packingStatus || "Pending",
      o.packedFlags || "",
      "",
      "",
      "",
      o.remarks || ""
    ]);
    return ContentService.createTextOutput(JSON.stringify({ status: "ok" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // Match row by Date & Time + Customer Name
  for (let i = 1; i < data.length; i++) {
    const matchTime = String(data[i][0]) === String(payload.createdAt);
    const matchName = String(data[i][1]) === String(payload.matchCustomerName || payload.customerName);
    if (matchTime && matchName) {
      const rowNum = i + 1;
      if (payload.action === "editOrder") {
        const o = payload.order;
        sheet.getRange(rowNum, 2).setValue(o.customerName);
        sheet.getRange(rowNum, 3).setValue(o.contactNo);
        sheet.getRange(rowNum, 4).setValue(o.companyName || "");
        sheet.getRange(rowNum, 5).setValue(o.taxId || "");
        sheet.getRange(rowNum, 6).setValue(o.address);
        sheet.getRange(rowNum, 7).setValue(o.items);
        sheet.getRange(rowNum, 8).setValue(o.itemCount || 1);
        sheet.getRange(rowNum, 11).setValue(o.packedFlags || "");
      } else if (payload.action === "update") {
        sheet.getRange(rowNum, 10).setValue(payload.packingStatus);
        sheet.getRange(rowNum, 11).setValue(payload.packedFlags || "");
        sheet.getRange(rowNum, 12).setValue(payload.weightBox || "");
        sheet.getRange(rowNum, 13).setValue(payload.courierName || "");
        sheet.getRange(rowNum, 14).setValue(payload.trackingId || "");
        sheet.getRange(rowNum, 15).setValue(payload.remarks || "");
      }
      break;
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ status: "ok" }))
    .setMimeType(ContentService.MimeType.JSON);
}
