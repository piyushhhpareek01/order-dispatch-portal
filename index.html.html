const SHEET_NAME = "Orders";

function setupSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
  }
  if (sheet.getLastRow() === 0) {
    sheet.appendRow([
      "Order ID",
      "Date & Time",
      "Customer Name",
      "Contact No",
      "Address",
      "Items & Quantity",
      "Total Items",
      "Payment Status",
      "Packing Status",
      "Packed Checklist",
      "Remarks",
      "Weight / Box Size",
      "Courier Name",
      "Tracking ID / AWB"
    ]);
    sheet.getRange(1, 1, 1, 14).setFontWeight("bold");
  }
  return sheet;
}

function doGet(e) {
  const sheet = setupSheet();
  const data = sheet.getDataRange().getValues();
  const orders = [];
  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    if (!row[0]) continue;
    orders.push({
      id: String(row[0]),
      createdAt: String(row[1]),
      customerName: String(row[2]),
      contactNo: String(row[3]),
      address: String(row[4]),
      items: String(row[5]),
      itemCount: Number(row[6] || 1),
      paymentStatus: String(row[7] || "Success"),
      packingStatus: String(row[8] || "Pending Packing"),
      packedFlags: String(row[9] || ""),
      remarks: String(row[10] || ""),
      weightBox: String(row[11] || ""),
      courierName: String(row[12] || ""),
      trackingId: String(row[13] || "")
    });
  }
  return ContentService.createTextOutput(JSON.stringify({ status: "ok", orders: orders.reverse() }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const sheet = setupSheet();
  const payload = JSON.parse(e.postData.contents);

  if (payload.action === "create") {
    const o = payload.order;
    sheet.appendRow([
      o.id,
      o.createdAt,
      o.customerName,
      o.contactNo,
      o.address,
      o.items,
      o.itemCount || 1,
      "Success",
      "Pending Packing",
      o.packedFlags || "",
      o.remarks || "",
      "",
      "",
      ""
    ]);
    return ContentService.createTextOutput(JSON.stringify({ status: "ok" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  if (payload.action === "editOrder") {
    const o = payload.order;
    const data = sheet.getDataRange().getValues();
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]) === String(o.id)) {
        const rowNum = i + 1;
        sheet.getRange(rowNum, 3).setValue(o.customerName);
        sheet.getRange(rowNum, 4).setValue(o.contactNo);
        sheet.getRange(rowNum, 5).setValue(o.address);
        sheet.getRange(rowNum, 6).setValue(o.items);
        sheet.getRange(rowNum, 7).setValue(o.itemCount || 1);
        sheet.getRange(rowNum, 10).setValue(o.packedFlags || "");
        break;
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ status: "ok" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  if (payload.action === "update") {
    const data = sheet.getDataRange().getValues();
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]) === String(payload.id)) {
        const rowNum = i + 1;
        sheet.getRange(rowNum, 9).setValue(payload.packingStatus);
        sheet.getRange(rowNum, 10).setValue(payload.packedFlags || "");
        sheet.getRange(rowNum, 11).setValue(payload.remarks || "");
        sheet.getRange(rowNum, 12).setValue(payload.weightBox || "");
        sheet.getRange(rowNum, 13).setValue(payload.courierName || "");
        sheet.getRange(rowNum, 14).setValue(payload.trackingId || "");
        break;
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ status: "ok" }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
